## Basic Method Signatures

### Simple method with no parameters

```ruby
# typed: true
class User
  extend T::Sig

  sig { returns(String) }
  def full_name
    "#{first_name} #{last_name}"
  end
end
```

### Method with parameters

```ruby
sig { params(name: String, age: Integer).returns(String) }
def greet(name, age)
  "Hello #{name}, you are #{age} years old"
end
```

### Method with optional parameters

```ruby
sig { params(name: String, age: T.nilable(Integer)).returns(String) }
def greet(name, age = nil)
  age ? "Hello #{name}, #{age}" : "Hello #{name}"
end
```

### Method with keyword arguments

```ruby
sig { params(name: String, email: String, admin: T::Boolean).returns(User) }
def create_user(name:, email:, admin: false)
  User.create!(name: name, email: email, admin: admin)
end
```

### Void methods (no return value)

```ruby
sig { params(message: String).void }
def log(message)
  Rails.logger.info(message)
end
```

### Class methods

```ruby
sig { params(id: String).returns(T.nilable(User)) }
def self.find_by_external_id(id)
  find_by(external_id: id)
end
```

## ActiveRecord Associations

### belongs_to (non-optional)

```ruby
# typed: true
class Exchange < ApplicationRecord
  extend T::Sig

  # Non-optional association (always present)
  sig { returns(Register) }
  def register
    super
  end

  # Or use attr_reader for cleaner syntax
  sig { returns(Register) }
  attr_reader :register
end
```

### belongs_to (optional)

```ruby
sig { returns(T.nilable(ImportOrigin)) }
def import_origin
  super
end
```

### has_many

```ruby
sig { returns(ActiveRecord::Associations::CollectionProxy[Exchange]) }
def exchanges
  super
end
```

### has_one

```ruby
sig { returns(T.nilable(Profile)) }
def profile
  super
end
```

### has_many through

```ruby
sig { returns(ActiveRecord::Associations::CollectionProxy[Tag]) }
def tags
  super
end
```

### Declaring associations with proper types

```ruby
# typed: true
class Book < ApplicationRecord
  extend T::Sig

  # belongs_to (non-optional)
  sig { returns(User) }
  attr_reader :owner

  # has_many
  sig { returns(ActiveRecord::Associations::CollectionProxy[Register]) }
  def registers; end

  sig { returns(ActiveRecord::Associations::CollectionProxy[Exchange]) }
  def exchanges; end

  sig { returns(ActiveRecord::Associations::CollectionProxy[Reminder]) }
  def reminders; end
end
```

## Scopes and Class Methods

### Scopes

Scopes are typically left untyped or use `T.untyped` because they return complex relation objects:

```ruby
# typed: true
class Register < ApplicationRecord
  extend T::Sig

  # Scopes typically don't need signatures (Tapioca generates them)
  scope :accounts, -> { where(register_type: "account") }
  scope :categories, -> { where(register_type: "category") }
end
```

### Query methods

```ruby
sig { params(status: String).returns(ActiveRecord::Relation) }
def self.with_status(status)
  where(status: status)
end

sig { params(book_id: String).returns(ActiveRecord::Relation) }
def self.for_book(book_id)
  where(book_id: book_id)
end
```

## Enums

Enums are auto-typed by Tapioca, but you can add signatures for custom methods:

```ruby
# typed: true
class Exchange < ApplicationRecord
  extend T::Sig

  enum :status, {
    draft: "draft",
    pending: "pending",
    cleared: "cleared"
  }

  # Enum predicates are auto-generated by Tapioca
  # draft?, pending?, cleared? are all typed automatically

  sig { returns(T::Boolean) }
  def finalized?
    !draft?
  end
end
```

## Union and Nilable Types

### Nilable types

```ruby
sig { returns(T.nilable(String)) }
def optional_description
  description.presence
end

sig { params(value: T.nilable(Integer)).void }
def set_amount(value)
  @amount = value
end
```

### Union types (multiple possible types)

```ruby
sig { params(id: T.any(String, Integer)).returns(T.nilable(User)) }
def find_user(id)
  User.find_by(id: id) || User.find_by(external_id: id.to_s)
end

sig { returns(T.any(Money::Currency, String)) }
def currency_representation
  currency_object || currency_code
end
```

### Type assertions (when you know better than Sorbet)

```ruby
sig { returns(User) }
def current_user
  # T.must asserts that the value is not nil
  T.must(@current_user)
end

sig { returns(String) }
def user_name
  # T.cast forces a type (use sparingly!)
  T.cast(session[:name], String)
end
```

## Hash Types

### Simple hash with string keys

```ruby
sig { returns(T::Hash[String, T.untyped]) }
def metadata
  { "created_at" => Time.now, "version" => 1 }
end
```

### Hash with symbol keys

```ruby
sig { params(params: T::Hash[Symbol, T.untyped]).returns(User) }
def create_from_params(params)
  User.create!(params)
end
```

### Typed hash values

```ruby
sig { returns(T::Hash[String, Integer]) }
def status_counts
  { "pending" => 5, "cleared" => 10 }
end

sig { returns(T::Hash[Symbol, String]) }
def config
  { host: "localhost", port: "3000" }
end
```

### Nested hashes

```ruby
sig { returns(T::Hash[String, T::Hash[String, T.untyped]]) }
def nested_data
  { "user" => { "name" => "John", "age" => 30 } }
end
```

## Generic Types

### Arrays

```ruby
sig { returns(T::Array[String]) }
def tag_names
  tags.map(&:name)
end

sig { params(ids: T::Array[String]).returns(T::Array[Exchange]) }
def find_exchanges(ids)
  Exchange.where(id: ids)
end
```

### Sets

```ruby
sig { returns(T::Set[String]) }
def unique_currencies
  Set.new(registers.map(&:currency_iso_code))
end
```

### Ranges

```ruby
sig { returns(T::Range[Date]) }
def date_range
  start_date..end_date
end
```

## Concerns and Modules

### Module with instance methods

```ruby
# typed: true
module Taggable
  extend T::Sig
  extend ActiveSupport::Concern

  sig { returns(T::Array[String]) }
  def tag_names
    tags.map(&:name)
  end

  sig { params(name: String).void }
  def add_tag(name)
    tags << Tag.find_or_create_by(name: name)
  end
end
```

### Module with class methods

```ruby
# typed: true
module Importable
  extend T::Sig
  extend ActiveSupport::Concern

  included do
    belongs_to :import_origin, optional: true
  end

  class_methods do
    sig { params(origin_id: String).returns(ActiveRecord::Relation) }
    def imported_from(origin_id)
      where(import_origin_id: origin_id)
    end
  end
end
```

### Module with DSL (like Currencyable)

For DSLs with metaprogramming, create a manual RBI file (see `sorbet/rbi/manual/currencyable.rbi`).

## Rails Controllers

Controllers can be typed, but often use `T.untyped` for params:

```ruby
# typed: true
class BooksController < ApplicationController
  extend T::Sig

  sig { void }
  def index
    @books = current_user.books
  end

  sig { void }
  def show
    @book = T.must(current_user.books.find(params[:id]))
  end

  private

  sig { returns(T.untyped) }
  def book_params
    params.require(:book).permit(:name, :default_currency_iso_code)
  end
end
```

## GraphQL Types and Mutations

### GraphQL Object Types

```ruby
# typed: true
module Types
  class BookType < Types::BaseObject
    extend T::Sig

    field :id, ID, null: false
    field :name, String, null: false
    field :default_currency, String, null: false

    # Custom resolver method
    sig { returns(String) }
    def default_currency
      object.default_currency.iso_code
    end

    # Method with authorization
    sig { returns(T::Array[Types::RegisterType]) }
    def registers
      authorize(object, :show?)
      object.registers
    end
  end
end
```

### GraphQL Mutations

```ruby
# typed: true
module Mutations
  class CreateBook < BaseMutation
    extend T::Sig

    argument :name, String, required: true
    argument :default_currency_iso_code, String, required: true

    field :book, Types::BookType, null: true
    field :errors, [String], null: false

    sig do
      params(
        name: String,
        default_currency_iso_code: String
      ).returns(T::Hash[Symbol, T.untyped])
    end
    def resolve(name:, default_currency_iso_code:)
      book = current_user.books.create(
        name: name,
        default_currency_iso_code: default_currency_iso_code
      )

      if book.persisted?
        { book: book, errors: [] }
      else
        { book: nil, errors: book.errors.full_messages }
      end
    end
  end
end
```

### GraphQL Input Types

```ruby
# typed: true
module Types
  class BookInputType < Types::BaseInputObject
    extend T::Sig

    argument :name, String, required: true
    argument :default_currency_iso_code, String, required: true
    argument :description, String, required: false

    # Input types typically don't need custom methods
  end
end
```

## Advanced Patterns

### Type Aliases

```ruby
# typed: true
module Types
  CurrencyCode = T.type_alias { String }
  BookID = T.type_alias { String }
  OptionalString = T.type_alias { T.nilable(String) }
end

class Book
  extend T::Sig

  sig { returns(Types::CurrencyCode) }
  def currency_code
    default_currency.iso_code
  end
end
```

### Abstract Classes

```ruby
# typed: true
class BaseService
  extend T::Sig
  extend T::Helpers

  abstract!

  sig { abstract.returns(T.untyped) }
  def call; end
end

class CreateBookService < BaseService
  sig { override.returns(Book) }
  def call
    Book.create!(name: "New Book")
  end
end
```

### Interfaces

```ruby
# typed: true
module Exportable
  extend T::Sig
  extend T::Helpers

  interface!

  sig { abstract.returns(String) }
  def to_csv; end

  sig { abstract.returns(String) }
  def to_json; end
end

class Book
  extend T::Sig
  include Exportable

  sig { override.returns(String) }
  def to_csv
    "#{name},#{default_currency.iso_code}"
  end

  sig { override.returns(String) }
  def to_json
    { name: name, currency: default_currency.iso_code }.to_json
  end
end
```

## Best Practices

1. **Start with `# typed: true`** - Don't jump to strict mode immediately
2. **Use `T.nilable` liberally** - Better to be explicit about nil
3. **Avoid `T.untyped`** - Use it only when truly necessary
4. **Let Tapioca handle associations** - Don't manually type ActiveRecord methods unless needed
5. **Use `sig` for all public methods** - Private methods can optionally skip signatures
6. **Use `T.must` sparingly** - Only when you're certain a value isn't nil
7. **Create manual RBIs for gems** - Don't pollute application code with gem type definitions
8. **Use type aliases for complex types** - Makes signatures more readable

## Common Mistakes

### ❌ Don't wrap non-nilable return types in T.nilable

```ruby
# Bad
sig { returns(T.nilable(String)) }
def name
  "John"  # This can never be nil
end

# Good
sig { returns(String) }
def name
  "John"
end
```

### ❌ Don't use T.untyped when you know the type

```ruby
# Bad
sig { returns(T.untyped) }
def user_ids
  users.map(&:id)
end

# Good
sig { returns(T::Array[String]) }
def user_ids
  users.map(&:id)
end
```

### ❌ Don't add signatures to private methods unnecessarily

```ruby
# Acceptable (but not required for private methods)
private

def helper_method
  # ...
end

# Also acceptable
private

sig { returns(String) }
def helper_method
  # ...
end
```

### ❌ Don't type test files at strict levels

```ruby
# Bad
# typed: strict  # Tests don't need strict mode

# Good
# typed: false   # Or true if you want basic checking
```
